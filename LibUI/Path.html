def self.open(fill_mode = DrawFillModeWinding)
  path = new(fill_mode)
  begin
    yield path
  ensure
    path.close
  end
end

def close
  return if @pointer.nil?

  LibUI.draw_free_path(@pointer)
  @pointer = nil
  nil
end
